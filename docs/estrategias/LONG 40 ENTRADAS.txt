//@version=6
strategy("Estrategia Corta 1M - Con Alertas Telegram", overlay=true, initial_capital=36000, default_qty_value=300000, default_qty_type=strategy.cash, margin_long=0.10)

// --- INPUTS DE INDICADORES ---
// Configuración de MACD del marco de tiempo actual
fastLength = input.int(12, "MACD Fast Length")
slowLength = input.int(26, "MACD Slow Length")
signalLength = input.int(9, "MACD Signal Length")

// Configuración de RSI del marco de tiempo actual (1 minuto)
rsiLength = input.int(14, "RSI Length (1m)")
rsiThreshold = input.float(50.0, "RSI Buy Threshold (1m)", minval=0, maxval=100)

// Configuración de RSI del marco de tiempo superior (5 minutos)
rsiLengthMTF = input.int(14, "RSI Length (5m)")
rsiThresholdMTF = input.float(40.0, "RSI Threshold (5m)", minval=0, maxval=100)

// Configuración de Stop Loss en USD
stopLossUSD = input.float(245, "Stop Loss en USD")

// Configuración de Trailing Take Profit
trailingActivateUSD = input.float(185, "Trailing TP Activate (USD)", minval=0)
trailingStopUSD = input.float(50, "Trailing TP Distance (USD)", minval=0)

// Configuración para el Stop Loss al Nivel de Ganancia
// Esta variable define la ganancia en USD que la posición debe alcanzar para que el
// Stop Loss se mueva a un nuevo nivel de protección (+90 USD del precio de entrada).
profitTargetForSLMove = input.float(300, "Objetivo de Ganancia para Mover SL (USD)", minval=0, tooltip="El SL se moverá a +90 USD del precio de entrada una vez que la ganancia sea de X USD")

// --- CÁLCULOS DE INDICADORES ---
// Cálculo de MACD del marco de tiempo actual
[macdLine, signalLine, histLine] = ta.macd(close, fastLength, slowLength, signalLength)

// Cálculo de RSI del marco de tiempo actual (1 minuto)
rsiValue = ta.rsi(close, rsiLength)

// Cálculo de RSI de un marco de tiempo superior (5 minutos)
rsiValueMTF = request.security(syminfo.tickerid, "5", ta.rsi(close, rsiLengthMTF))

// --- LÓGICA DE ENTRADA FINAL ---
// Regla 1: La línea del MACD del marco de tiempo actual cruza por encima de la línea de señal.
bool macdCrossoverCondition = ta.crossover(macdLine, signalLine)

// Regla 2: El filtro de MACD se cumple si es menor que -30 O si está entre -10 y -5.
bool macdFilterCondition = macdLine < -30 or (macdLine > -10 and macdLine < -5)

// Regla 3: El valor del RSI del marco de tiempo actual es mayor que el umbral de compra.
bool rsiFilterCondition = rsiValue > rsiThreshold

// Regla 4: El valor del RSI de 5 minutos es mayor o igual a 40.
bool mtfRsiFilterCondition = rsiValueMTF >= rsiThresholdMTF

// Condición combinada: Todas las reglas DEBEN ser verdaderas para entrar.
bool longEntryCondition = macdCrossoverCondition and macdFilterCondition and rsiFilterCondition and mtfRsiFilterCondition

// --- VARIABLES PARA LA GESTIÓN DE SALIDAS DINÁMICAS ---
// Usamos 'var' para mantener el valor de esta variable a lo largo de las barras
var float dynamicSL = na
var float prev_position_size = 0.0 // CORRECCIÓN: Declaración de 'prev_position_size' como var global

// --- EJECUCIÓN DE LA ESTRATEGIA ---
// Mensaje de alerta para la entrada LONG que Cornix puede interpretar
string entryAlertMessage = "BUY " + syminfo.tickerid + " entry_price=" + str.tostring(close) + " percent=100%"

// Entrar en una posición larga cuando se cumplen las condiciones
if (longEntryCondition)
    strategy.entry("Long", strategy.long) // Eliminamos alert_message de strategy.entry
    // Generamos la alerta directamente con alert()
    alert(entryAlertMessage, alert.freq_once_per_bar_close)

// Lógica de gestión de la posición
if (strategy.position_size > 0)
    // Precio de entrada
    float entryPrice = strategy.opentrades.entry_price(0)
    
    // Nivel de ganancia en el que se activa el nuevo SL
    float breakEvenProfitLevel = entryPrice + profitTargetForSLMove

    // Nivel del nuevo SL (+90 USD del precio de entrada)
    float newSLPrice = entryPrice + 90
    
    // Inicializar o actualizar el SL dinámico en cada barra
    if (na(dynamicSL) or close < entryPrice) // Reiniciar si no está inicializado o el precio cae por debajo del punto de entrada
        dynamicSL := entryPrice - stopLossUSD
        
    // Lógica para mover el SL al nuevo nivel de ganancia
    // Si el precio actual supera el nivel de ganancia objetivo Y el SL aún no ha sido movido,
    // lo movemos a 90 USD por encima del precio de entrada.
    if (close >= breakEvenProfitLevel and dynamicSL < newSLPrice)
        dynamicSL := newSLPrice
    
    // Calcular los puntos de trailing TP
    float trailActivatePoints = trailingActivateUSD / syminfo.mintick
    float trailStopPoints = trailingStopUSD / syminfo.mintick

    // Mensaje de alerta para la salida (cierre de la posición LONG) que Cornix puede interpretar
    string exitAlertMessage = "SELL " + syminfo.tickerid + " exit_price=" + str.tostring(close)

    // Ejecutamos la salida para el backtesting
    strategy.exit("Exit", from_entry="Long", stop=dynamicSL, trail_offset=trailActivatePoints, trail_points=trailStopPoints)

    // Si la posición se cierra en la barra actual (verificando si strategy.position_size es 0 y lo era antes)
    // Nota: Esto requiere una variable 'var' para rastrear el tamaño de la posición anterior.
    if (strategy.position_size == 0 and prev_position_size > 0)
        alert(exitAlertMessage, alert.freq_once_per_bar_close)
    prev_position_size := strategy.position_size
else if (strategy.position_size == 0) // Reset prev_position_size if no trade
    prev_position_size := 0.0


// --- PLOTS PARA VISUALIZACIÓN ---
// Trazar una flecha verde para las señales de entrada
plotshape(series=longEntryCondition, title="Long Entry", style=shape.arrowup, location=location.belowbar, color=color.new(color.green, 0), size=size.normal)
