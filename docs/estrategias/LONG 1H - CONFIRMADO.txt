//@version=6
strategy("Estrategia MACD 1H Confirmada", overlay=true, initial_capital=30000, default_qty_value=300000, default_qty_type=strategy.cash, margin_long=0.10)

// --- INPUTS DE USUARIO ---
sendExitAlerts = input.bool(true, "Enviar Alertas de Salida", tooltip="Activa para enviar alertas cuando la posición se cierra.")
useTrailingTP = input.bool(true, "Usar Trailing TP?", tooltip="Activa para usar Trailing TP. Si no está activo, se usa el TP fijo.")

// Códigos de Alerta para BingX
alertEntryLongCode = "ENTER-LONG_BINANCE_BTCUSDT_1H-LONG_1M_c5a64cb2238d28970fde6dc5"
alertEntryShortCode = "ENTER-SHORT_BINANCE_BTCUSDT_1H-LONG_1M_c5a64cb2238d28970fde6dc5"
alertExitCode = "EXIT-LONG_BINANCE_BTCUSDT_1H-LONG_1M_c5a64cb2238d28970fde6dc5"

// --- CONFIGURACIÓN DE INDICADORES ---
// MACD (1H)
macdFastLength = input.int(12, "MACD Fast Length")
macdSlowLength = input.int(26, "MACD Slow Length")
macdSignalLength = input.int(9, "MACD Signal Length")

// --- CONFIGURACIÓN DE GESTIÓN DE RIESGO EN PORCENTAJES ---
stopLossPercent = input.float(0.18, "Stop Loss (%)", minval=0.01)
fixedTakeProfitPercent = input.float(1.0, "Take Profit Fijo (%)", minval=0.01)
trailingActivatePercent = input.float(1.0, "Trailing TP Activate (%)", minval=0.01)
trailingStopPercent = input.float(0.05, "Trailing TP Distance (%)", minval=0.01)

// --- CÁLCULOS DE INDICADORES ---
// MACD de 1 hora
[macdLine_1h, signalLine_1h, histLine_1h] = request.security(syminfo.tickerid, "60", ta.macd(close, macdFastLength, macdSlowLength, macdSignalLength))

// --- LÓGICA DE ENTRADA FINAL ---
// Condición para Long: Cruce alcista confirmado (histograma positivo)
bool longEntryCondition = ta.crossover(macdLine_1h, signalLine_1h) and histLine_1h > 0

// Condición para Short: Cruce bajista confirmado (histograma negativo)
bool shortEntryCondition = ta.crossunder(macdLine_1h, signalLine_1h) and histLine_1h < 0

// --- EJECUCIÓN DE LA ESTRATEGIA ---
// Lógica de entrada
if (longEntryCondition)
    strategy.entry("Long", strategy.long)
    alert(alertEntryLongCode)

if (shortEntryCondition)
    strategy.entry("Short", strategy.short)
    alert(alertEntryShortCode)

// Lógica de gestión de salida
if (strategy.position_size > 0)
    float entryPrice = strategy.opentrades.entry_price(0)
    float stopLossPrice = entryPrice * (1 - (stopLossPercent / 100))
    float fixedTakeProfitPrice = entryPrice * (1 + (fixedTakeProfitPercent / 100))
    
    if (useTrailingTP)
        float trailActivatePrice = entryPrice * (1 + (trailingActivatePercent / 100))
        float trailStopPoints = (entryPrice * (trailingStopPercent / 100)) / syminfo.mintick
        strategy.exit("ExitLong", from_entry="Long", stop=stopLossPrice, trail_price=trailActivatePrice, trail_offset=trailStopPoints)
    else
        strategy.exit("ExitLong", from_entry="Long", stop=stopLossPrice, limit=fixedTakeProfitPrice)
else if (strategy.position_size < 0)
    float entryPrice = strategy.opentrades.entry_price(0)
    float stopLossPrice = entryPrice * (1 + (stopLossPercent / 100))
    float fixedTakeProfitPrice = entryPrice * (1 - (fixedTakeProfitPercent / 100))

    if (useTrailingTP)
        float trailActivatePrice = entryPrice * (1 - (trailingActivatePercent / 100))
        float trailStopPoints = (entryPrice * (trailingStopPercent / 100)) / syminfo.mintick
        strategy.exit("ExitShort", from_entry="Short", stop=stopLossPrice, trail_price=trailActivatePrice, trail_offset=trailStopPoints)
    else
        strategy.exit("ExitShort", from_entry="Short", stop=stopLossPrice, limit=fixedTakeProfitPrice)

// Lógica de alerta de salida separada
if (strategy.position_size == 0 and (strategy.position_size[1] > 0 or strategy.position_size[1] < 0))
    if (sendExitAlerts)
        alert(alertExitCode)

// --- VISUALIZACIÓN ---
plotshape(series=longEntryCondition, title="Long Entry", style=shape.arrowup, location=location.belowbar, color=color.new(color.green, 0), size=size.normal)
plotshape(series=shortEntryCondition, title="Short Entry", style=shape.arrowdown, location=location.abovebar, color=color.new(color.red, 0), size=size.normal)
