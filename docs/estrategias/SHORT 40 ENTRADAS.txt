//@version=6
strategy("Estrategia Corta 1M (Invertida)", overlay=true, initial_capital=36000, default_qty_value=360000, default_qty_type=strategy.cash, margin_short=0.10) // CAMBIO: Nombre y margin_short

// --- INPUTS DE INDICADORES ---
// Configuración de MACD del marco de tiempo actual
fastLength = input.int(12, "MACD Fast Length")
slowLength = input.int(26, "MACD Slow Length")
signalLength = input.int(9, "MACD Signal Length")

// Configuración de RSI del marco de tiempo actual (1 minuto)
rsiLength = input.int(14, "RSI Length (1m)")
rsiThreshold = input.float(50.0, "RSI Sell Threshold (1m)", minval=0, maxval=100) // CAMBIO: Texto a Sell Threshold

// Configuración de RSI del marco de tiempo superior (5 minutos)
rsiLengthMTF = input.int(14, "RSI Length (5m)")
rsiThresholdMTF = input.float(40.0, "RSI Threshold (5m)", minval=0, maxval=100) // CAMBIO: Umbral MTF

// Configuración de Stop Loss en USD (PARA SHORT, es por encima del precio de entrada)
stopLossUSD = input.float(245, "Stop Loss en USD")

// Configuración de Trailing Take Profit (PARA SHORT, es por debajo del precio de entrada)
trailingActivateUSD = input.float(185, "Trailing TP Activate (USD)", minval=0)
trailingStopUSD = input.float(50, "Trailing TP Distance (USD)", minval=0)

// Configuración para el Stop Loss al Nivel de Ganancia (PARA SHORT)
// Esta variable define la ganancia en USD que la posición debe alcanzar para que el
// Stop Loss se mueva a un nuevo nivel de protección (-90 USD del precio de entrada).
profitTargetForSLMove = input.float(145, "Objetivo de Ganancia para Mover SL (USD)", minval=0, tooltip="El SL se moverá a -90 USD del precio de entrada una vez que la ganancia sea de X USD")

// --- CÁLCULOS DE INDICADORES ---
// Cálculo de MACD del marco de tiempo actual
[macdLine, signalLine, histLine] = ta.macd(close, fastLength, slowLength, signalLength)

// Cálculo de RSI del marco de tiempo actual (1 minuto)
rsiValue = ta.rsi(close, rsiLength)

// Cálculo de RSI de un marco de tiempo superior (5 minutos)
rsiValueMTF = request.security(syminfo.tickerid, "5", ta.rsi(close, rsiLengthMTF))

// --- LÓGICA DE ENTRADA FINAL (INVERTIDA PARA SHORT) ---
// Regla 1: La línea del MACD del marco de tiempo actual cruza por debajo de la línea de señal.
bool macdCrossunderCondition = ta.crossunder(macdLine, signalLine) // CAMBIO: Crossover a Crossunder

// Regla 2: El filtro de MACD se cumple si es mayor que 30 O si está entre 5 y 10.
bool macdFilterCondition = macdLine > 30 or (macdLine < 10 and macdLine > 5) // CAMBIO: Rango invertido para territorio positivo

// Regla 3: El valor del RSI del marco de tiempo actual es menor que el umbral de venta.
bool rsiFilterCondition = rsiValue < rsiThreshold // CAMBIO: Mayor que a Menor que

// Regla 4: El valor del RSI de 5 minutos es menor o igual a 40.
bool mtfRsiFilterCondition = rsiValueMTF <= rsiThresholdMTF // CAMBIO: Mayor o igual a Menor o igual

// Condición combinada: Todas las reglas DEBEN ser verdaderas para entrar en SHORT.
bool shortEntryCondition = macdCrossunderCondition and macdFilterCondition and rsiFilterCondition and mtfRsiFilterCondition // CAMBIO: longEntry a shortEntry

// --- VARIABLES PARA LA GESTIÓN DE SALIDAS DINÁMICAS ---
// Usamos 'var' para mantener el valor de esta variable a lo largo de las barras
var float dynamicSL = na

// --- EJECUCIÓN DE LA ESTRATEGIA ---
// Entrar en una posición SHORT cuando se cumplen las condiciones
if (shortEntryCondition)
    strategy.entry("Short", strategy.short) // CAMBIO: strategy.long a strategy.short

// Lógica de gestión de la posición SHORT
if (strategy.position_size < 0) // CAMBIO: > 0 a < 0 para posición SHORT
    // Precio de entrada
    float entryPrice = strategy.opentrades.entry_price(0)
    
    // Nivel de ganancia en el que se activa el nuevo SL (PARA SHORT: precio baja)
    float breakEvenProfitLevel = entryPrice - profitTargetForSLMove // CAMBIO: + a -

    // Nivel del nuevo SL (-90 USD del precio de entrada para SHORT)
    float newSLPrice = entryPrice - 90 // CAMBIO: + a -
    
    // Inicializar o actualizar el SL dinámico en cada barra
    if (na(dynamicSL) or close > entryPrice) // CAMBIO: close < entryPrice a close > entryPrice (para SHORT)
        dynamicSL := entryPrice + stopLossUSD // CAMBIO: - a + (SL inicial por encima de la entrada para SHORT)
        
    // Lógica para mover el SL al nuevo nivel de ganancia (PARA SHORT)
    // Si el precio actual está por debajo del nivel de ganancia objetivo Y el SL aún no ha sido movido,
    // lo movemos a 90 USD por debajo del precio de entrada.
    if (close <= breakEvenProfitLevel and dynamicSL > newSLPrice) // CAMBIO: >= a <=, < a >
        dynamicSL := newSLPrice
    
    // Calcular los puntos de trailing TP
    float trailActivatePoints = trailingActivateUSD / syminfo.mintick
    float trailStopPoints = trailingStopUSD / syminfo.mintick

    // Salir de la posición usando el SL dinámico y el Trailing TP
    strategy.exit("Exit", from_entry="Short", stop=dynamicSL, trail_offset=trailActivatePoints, trail_points=trailStopPoints) // CAMBIO: from_entry="Long" a "Short"

// --- PLOTS PARA VISUALIZACIÓN ---
// Trazar una flecha roja para las señales de entrada SHORT
plotshape(series=shortEntryCondition, title="Short Entry", style=shape.arrowdown, location=location.abovebar, color=color.new(color.red, 0), size=size.normal) // CAMBIO: Flecha y color
