//@version=6
strategy("Estrategia Corta 1M - 20 Entradas - Ali (Sincronizada)", overlay=true, initial_capital=30000, default_qty_value=300000, default_qty_type=strategy.cash, margin_long=0.10, process_orders_on_close=true)

// --- INPUTS DE INDICADORES ---
sendExitAlerts = input.bool(true, "Enviar Alertas de Salida", tooltip="Activa esta opción si quieres que TradingView envíe las alertas de salida al bot.")
fastLength = input.int(12, "MACD Fast Length")
slowLength = input.int(26, "MACD Slow Length")
signalLength = input.int(9, "MACD Signal Length")
rsiLength = input.int(14, "RSI Length (1m)")
rsiThreshold = input.float(50.0, "RSI Buy Threshold (1m)", minval=0, maxval=100)
rsiLengthMTF = input.int(14, "RSI Length (5m)")
rsiThresholdMTF = input.float(40.0, "RSI Threshold (5m)", minval=0, maxval=100)

// --- OPCIONES PARA TAKE PROFIT Y STOP LOSS EN PORCENTAJE ---
stopLossPercent = input.float(1, "Stop Loss en %", tooltip="Porcentaje de pérdida máxima desde el precio de entrada.", minval=0.01)
useTrailingTP = input.bool(true, "Usar Trailing TP?")
trailingActivatePercent = input.float(0.45, "Trailing TP Activate (%)", minval=0.01, tooltip="Porcentaje de ganancia para activar el trailing stop.")
trailingStopPercent = input.float(0.05, "Trailing TP Distance (%)", minval=0.01, tooltip="Porcentaje de distancia que el trailing stop mantiene del precio.")
fixedTakeProfitPercent = input.float(0.5, "Take Profit Fijo en %")

// Códigos de alerta para el bot
commentEntry = "ENTER-LONG_BINANCE_BTCUSDT_ALI_1M_c5e14cb2ab8d28970f376d68"
commentExit = "EXIT-LONG_BINANCE_BTCUSDT_ALI_1M_c55a4cb2878d28970f926d3d"
macdMtfThreshold = input.float(60.0, "MACD 5m No-Entry Threshold")

// --- CÁLCULOS DE INDICADORES ---
[macdLine, signalLine, histLine] = ta.macd(close, fastLength, slowLength, signalLength)
[macdMtfLine, signalMtfLine, histMtfLine] = request.security(syminfo.tickerid, "5", ta.macd(close, fastLength, slowLength, signalLength))
rsiValue = ta.rsi(close, rsiLength)
rsiValueMTF = request.security(syminfo.tickerid, "5", ta.rsi(close, rsiLengthMTF))

// Lógica de sincronización de la barra de 5 minutos
bool isNewBar5m = request.security(syminfo.tickerid, "5", barstate.isconfirmed)

// --- LÓGICA DE ENTRADA FINAL ---
bool macdCrossoverCondition = ta.crossover(macdLine, signalLine)
bool macdFilterCondition = macdLine < -30 or (macdLine > -10 and macdLine < -5)
bool rsiFilterCondition = rsiValue > rsiThreshold
bool mtfRsiFilterCondition = rsiValueMTF >= rsiThresholdMTF
bool mtfMacdFilterCondition = macdMtfLine < macdMtfThreshold
// La condición de entrada completa requiere el cierre de la barra de 5m
bool longEntryCondition = isNewBar5m and macdCrossoverCondition and macdFilterCondition and rsiFilterCondition and mtfRsiFilterCondition and mtfMacdFilterCondition

// --- EJECUCIÓN DE LA ESTRATEGIA ---
// Lógica para la entrada (con alerta dedicada)
if (longEntryCondition)
    strategy.entry("Long", strategy.long)
    alert(commentEntry)

// Lógica de gestión de la posición
if (strategy.position_size > 0)
    float entryPrice = strategy.opentrades.entry_price(0)
    
    // Convertir porcentajes a precios o ticks
    float stopLossPrice = entryPrice * (1 - stopLossPercent / 100)
    float fixedTakeProfitPrice = entryPrice * (1 + fixedTakeProfitPercent / 100)
    float trailActivatePrice = entryPrice * (1 + trailingActivatePercent / 100)
    float trailStopPoints = (entryPrice * (trailingStopPercent / 100)) / syminfo.mintick

    // Salida con SL y TP seleccionable (fijo o trailing)
    if (useTrailingTP)
        strategy.exit("Exit", from_entry="Long", stop=stopLossPrice, trail_price=trailActivatePrice, trail_offset=trailStopPoints)
    else
        strategy.exit("Exit", from_entry="Long", stop=stopLossPrice, limit=fixedTakeProfitPrice)
    
// Lógica de alerta de salida separada
if (strategy.position_size == 0 and strategy.position_size[1] > 0)
    if (sendExitAlerts)
        alert(commentExit)

// --- LABELS Y PLOTS ---
if (strategy.position_size > 0 and strategy.position_size[1] == 0)
    label.new(bar_index, low, "ENTRADA", xloc.bar_index, yloc.belowbar, color.new(color.green, 10), textcolor=color.white, style=label.style_label_up)

if (strategy.position_size == 0 and strategy.position_size[1] > 0)
    label.new(bar_index[1], high[1], "SALIDA", xloc.bar_index, yloc.abovebar, color.new(color.red, 10), textcolor=color.white, style=label.style_label_down)
    
plotshape(series=longEntryCondition, title="Long Entry", style=shape.arrowup, location=location.belowbar, color=color.new(color.green, 0), size=size.normal)
plotshape(series=mtfMacdFilterCondition, title="MACD 5m No-Entry", style=shape.diamond, location=location.top, color=color.new(color.blue, 70), size=size.tiny)
